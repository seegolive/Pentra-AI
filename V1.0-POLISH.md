@router.get("/findings/recent")
async def get_recent_findings(
    limit: int = Query(default=5, le=20),
    db: AsyncSession = Depends(get_db),

) -> list[FindingResponse]:

### B.3 — Error Handling UI

**Buat global error boundary:**

```typescript
// apps/web/src/components/ErrorBoundary.tsx

import React from "react";

interface State { hasError: boolean; error: Error | null }

export class ErrorBoundary extends React.Component<
  { children: React.ReactNode; fallback?: React.ReactNode },
  State
> {
  state: State = { hasError: false, error: null };

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback ?? (
        <div className="flex items-center justify-center h-40">
          <div className="text-center">
            <p className="text-red-400 text-sm font-medium">Something went wrong</p>
            <p className="text-slate-500 text-xs mt-1">
              {this.state.error?.message}
            </p>
            <button
              onClick={() => this.setState({ hasError: false, error: null })}
              className="mt-3 text-xs text-blue-500 hover:text-blue-400"
            >
              Try again
            </button>
          </div>
        </div>
      );
    }
    return this.props.children;
  }
}
```

// 3. Tambahkan timestamp yang lebih readable (relative time)
// 4. Group LLM_STREAM tokens menjadi satu block (bukan baris terpisah)
// 5. Highlight AWAITING_APPROVAL dengan border yang lebih jelas

// Fix LLM streaming — aggregate ke 1 block per node:
function groupLlmStream(events: FeedEvent[]): FeedEvent[] {
  const result: FeedEvent[] = [];
  let currentStream = "";

  for (const ev of events) {
    if (ev.type === "LLM_STREAM") {
      currentStream += ev.content ?? "";
    } else {
      if (currentStream) {
        result.push({ type: "LLM_STREAM", content: currentStream });
        currentStream = "";
      }
      result.push(ev);
    }
  }

  if (currentStream) {
    result.push({ type: "LLM_STREAM", content: currentStream });
  }

  return result;
}
